WITH student_loan AS (
        SELECT
            loan_id,
            MAX(federal_student_loan_flag) AS student_loan_flag
        FROM sandbox.student_loan_chime_flags
        GROUP BY 1
)

SELECT


 cd.account_number

     , cd.book_date
     , dd.calendar_week                                                            AS book_week
     , dd.calendar_quarter_name                                                    AS book_vintage
     , date(cd.application_received_dtm)                                           AS app_date
     --, bbvt.bb_wholesale                                                           AS bbvalue
    , bbvt.bb_value                                                           AS bbvalue
     ,cd.dealer_number                                                            AS dealer_number
     , cd.con_sales_price                                     AS sale_price
 -- , case when dru.riskdealergroup = 'FRN' and cdd.dealer_type_name = 'Independent' then 1 else 0 end AS frni_flag


   , case when dru.riskdealergroup = 'FRN' and cdd.dealer_type_name = 'Independent' then 1
        when dru.riskdealergroup = 'FRN' and cdd.dealer_type_name != 'Independent' then 0 else -1 end AS frni_flag



     , cd.con_amount_financed_back                                                 AS amt_financed
     , cd.con_cash_down_amt                                                        AS cash_down
     , cd.con_net_trade_amt                                                        AS tradein_value
     , cd.purchase_odometer                                                        AS mileage
     , cd.con_risk_model_score                                                      AS cd_model_score
     , CASE
           WHEN (cd.pb_monthly_income_gross_total + coalesce(cd.cb_monthly_income_gross_total, 0)) < 1 THEN 1
           ELSE cd.pb_monthly_income_gross_total + coalesce(cd.cb_monthly_income_gross_total, 0) END
                                                                                   AS total_income
     , cd.con_payment_front_amt / (CASE
                                       WHEN 12000 < total_income THEN 12000
                                       ELSE total_income END)                      AS pti
     , cd.purchase_make                                                           AS make
     , cd.purchase_year                                                     AS model_year
     , cd.purchase_vin                                                            AS vin
     ,case when dru. previous_riskdealergroup = '' then dru.riskdealergroup else dru.previous_riskdealergroup end                                                         AS lob
     , date_part(dayofweek, cd.application_received_dtm)

AS day_of_week
     , COALESCE(ccat.secured_credit_card,pcat.secured_credit_card, 0)                                       AS secured_credit_card

     ,COALESCE(ccat.chime_indicator, pcat.prov_chime, 0)                                       AS chime_indicator
     ,COALESCE(ccat.chime_indicator,cbcat.chime_indicator, pcat.prov_chime, 0)                    AS nonkmx_chime_indicator

     ,stu.student_loan_flag::int AS student_loan_flag

     ,case when cd.cb_customer_id is not null then 1.0000 else 0.0000 end as cb_flag





     ,case when fraud.fraud_adjustment > 0 then fraud.fraud_adjustment else 1.00 end as fraud_adjustment




     , ett.employment_type                                                         AS employment
     , cdd.pricing_specialty_dealer_name                                           AS specialty_dealer
     , cd.pb_primary_years_at_employer * 12 + cd.pb_primary_months_at_employer                                              AS employed_months
     , COALESCE(cd.pb_fico_score, 0)                                         AS fico_score
     , COALESCE(cast(ccat.vantage as int), 0)  AS vantage_score --COALESCE(cd.pb_vantage_score, 0)
     , cd.pb_current_state_code                                              AS state
     , lkat.app_type                                                               AS pull_type
     ,  COALESCE(ccat.dq_auto, pcat.dq_auto, 0)                                     AS existing_dq_count
     , CASE
           WHEN COALESCE(ccat.num_tradelines, pcat.num_tradelines) > 0 AND COALESCE(ccat.auth_tradelines, pcat.auth_tradelines) > 0
               THEN COALESCE(ccat.auth_tradelines, pcat.auth_tradelines)::float / COALESCE(ccat.num_tradelines, pcat.num_tradelines)::float
           ELSE 0.0 END                                                                                 AS pct_auth_tradelines
     , COALESCE(ccat.open_tradelines, pcat.open_tradelines)                                                                             AS open_tl
     , COALESCE(ccat.prev_chargeoff, pcat.prev_chargeoff,0)                                                                 AS prev_co_count
     , CASE
           WHEN dru.RiskDealerGroup in ('KMX', 'MCY') THEN dru.RiskDealerGroup
           WHEN dap."dealer level loss adjustment" = 'F' THEN 'E'
           ELSE dap."dealer level loss adjustment" END AS lob_or_bucket
     , case
           when cd.con_apr < .20 then 'A'
           when cd.con_apr < .24 then 'B'
           when cd.con_apr < .28 then 'C'
           else 'D'
    end                                                                            as apr_bucket
    , case
        when cd.application_received_date >= '2024-12-16'
               then 0
           when cd.application_received_date >= '2024-11-7' and cd.application_received_date < '2024-12-16' and
                lrn.ghostfile < 250 then 0
           else 1 end as fico_adjustment_flag

    , case when risk_model_version = 3.1 and dru.riskdealergroup = 'KMX' then 'MTN3.1' else '' end as mtn_model --blended KMX scores
    , fd.sentilink_score
    , pp_score



FROM edwnpi.los_deal_current_fact AS cd


        LEFT JOIN student_loan stu
                   ON stu.loan_id = cd.loan_id

         LEFT JOIN edwnpi.dealer_rollup_scd_current AS dru
                   ON dru.dealer_number = cd.dealer_number
         LEFT JOIN edwnpi.date_dim AS dd
                   ON cd.book_date = dd.calendar_date


         --LEFT JOIN #temp_blackbook_values AS bbvt
        LEFT join sandbox.rds_blackbook_rollup bbvt
                   ON cd.account_number = bbvt.account_number



         LEFT JOIN edwnpi.dealer_attributes_pivot AS dap
                   ON cd.dealer_number = dap.dealerid AND dap.datasourceid = 17
         LEFT JOIN edwnpi.crm_dealer_dim AS cdd
                   ON cdd.dealer_number = cd.dealer_number
                       AND cdd.current_version_flag = 1
         LEFT JOIN sandbox.temp_employment_type_ragu AS ett
                   ON ett.account_number = cd.account_number
         LEFT JOIN sandbox.kmx_los_new_sp lkat
                    ON lkat.loan_id = cd.loan_id
         LEFT JOIN sandbox.temp_prov_customer_credit_attributes_ragu AS pcat
                   ON pcat.customerid = cd.pb_customer_id
                   and cd.data_source_id = 17
         LEFT JOIN sandbox.temp_los_customer_credit_attributes_ragu AS ccat
                   ON ccat.customer_id = cd.pb_customer_id
                   and cd.data_source_id = 100
         --cb_chime_flag adj
         LEFT JOIN sandbox.temp_los_customer_credit_attributes_ragu AS cbcat
                    ON cbcat.customer_id = cd.cb_customer_id
         LEFT JOIN sandbox.loan_random_numbers lrn
                    ON lrn.loan_id = cd.loan_id
        LEFT JOIN sandbox.fraud_data_gs fd
                    ON fd.account_number = cd.account_number


        LEFT JOIN sandbox.temp_fraud_ragu as fraud
                    ON fraud.loan_id = cd.loan_id
--

  AND cd.data_source_name != 'SPARTAN'
