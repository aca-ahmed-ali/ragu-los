SELECT cd.account_number,
       cd.book_date,
       dd.calendar_week                      AS book_week,
       dd_app.calendar_week                  AS app_week,
       date(application_received_dtm) as app_date,
       case when dru. previous_riskdealergroup = '' then dru.riskdealergroup else dru.previous_riskdealergroup end AS lob,
       cd.purchase_year               AS car_year,
       cd.purchase_make                     AS car_make,
       cd.purchase_model                    AS car_model,
       cd.purchase_odometer                  AS mileage_orig,
       --bbvt.bb_wholesale                     AS bb_value,
         bbvt.bb_value  AS bb_value,
       cd.con_sales_price AS sale_price,
       --bbvt.car_class                        AS vehicle_class,
         bbvt.veh_class as vehicle_class,
       --bbvt.car_class_ext                    AS vehicle_class_extended,
       --bbvt.car_fuel                         AS fuel_type,
         bbvt.veh_fuel as fuel_type,
       cd.pb_primary_employer         AS job_company,
       case
           when cd.con_apr < .20 then 'A'
           when cd.con_apr < .24 then 'B'
           when cd.con_apr < .28 then 'C'
           else 'D'
           end                               as apr_bucket
FROM edwnpi.los_deal_current_fact cd
         LEFT JOIN edwnpi.crm_dealer_dim AS cdd
                   ON cd.dealer_number = cdd.dealer_number
                       AND cdd.current_version_flag = 1
         LEFT JOIN edwnpi.dealer_rollup_scd_current AS dru
                   ON cd.dealer_number = dru.dealer_number
         LEFT JOIN edwnpi.date_dim AS dd
                   ON cd.book_date = dd.calendar_date
         LEFT JOIN edwnpi.date_dim AS dd_app
                   ON date(application_received_dtm) = dd_app.calendar_date
         --LEFT JOIN #temp_blackbook_values AS bbvt
         LEFT JOIN sandbox.rds_blackbook_rollup AS bbvt
                   ON cd.account_number = bbvt.account_number
WHERE cd.account_number IS NOT NULL
  AND bbvt.bb_value > 0
  AND cd.data_source_name != 'SPARTAN'
  AND cd.book_date >= {min_book_date}
  --AND ((dru.riskdealergroup = 'KMX' and risk_model_version = 3.1 and app_date>='2024-12-01') or (dru.riskdealergroup = 'KMX' and app_date<'2024-12-01') or dru.riskdealergroup !='KMX')
