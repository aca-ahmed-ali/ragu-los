
DROP TABLE IF EXISTS #temp_employment_type;
SELECT cd.account_number,
       pb.job_title                          as pb_primary_job_title,
       employer_name                         as pb_primary_employer
        ,
       CASE
           WHEN pb_primary_job_title ILIKE '%server%' OR pb_primary_job_title ILIKE '%waiter%' OR
                pb_primary_job_title ILIKE '%waitress%' OR pb_primary_job_title ILIKE '%waitstaff%' OR
                pb_primary_job_title ILIKE '%bartender%'
               THEN 'waiter'
           WHEN pb_primary_job_title ILIKE '%painter%' OR pb_primary_job_title ILIKE '%landscap%' OR
                pb_primary_job_title ILIKE '%tempo%' OR pb_primary_job_title ILIKE '%manpower%' OR
                pb_primary_job_title ILIKE '%kelly%' OR pb_primary_job_title ILIKE '%spherion%' OR
                pb_primary_job_title ILIKE '%kforce%' OR pb_primary_job_title ilike '%staffing%' OR
                pb_primary_job_title ILIKE '%personnel%'
               OR pb_primary_employer ILIKE '%painter%' OR pb_primary_employer ILIKE '%landscap%' OR
                pb_primary_employer ILIKE '%tempo%' OR pb_primary_employer ILIKE '%manpower%' OR
                pb_primary_employer ILIKE '%kelly%' OR pb_primary_employer ILIKE '%spherion%' OR
                pb_primary_employer ILIKE '%kforce%' OR pb_primary_employer ILIKE '%staffing%' OR
                pb_primary_employer ILIKE '%personnel%'
               THEN 'seasonal'
           ELSE 'not seasonal or waiter' END AS employment_type,
       pb.months_at_employer,
       pb.years_at_employer
INTO #temp_employment_type
FROM sfs_booked_contracts_scd cd
         Left join (select loan_id,
                           application_received_date,
                           deal_detail_id,
                           source_document_id,
                           row_number()
                           over (partition by source_document_id order by application_received_date desc) rn
                    from odsnpi.sfs_deal_detail_scd
                    where current_version_flag = 1
                      and aspect = 'application') app
                   on cd.sfs_application_number = app.source_document_id and app.rn = 1
         left join odsnpi.sfs_deal_customer_scd pb_id
                   on app.deal_detail_id = pb_id.deal_detail_id and
                      pb_id.customer_role = 'Primary' and
                      pb_id.current_version_flag = 1
         left join odsnpi.sfs_deal_customer_scd cb_id
                   on app.deal_detail_id = cb_id.deal_detail_id and
                      cb_id.customer_role = 'Auxiliary' and
                      cb_id.current_version_flag = 1
         left join odsnpi.sfs_customer_employer_scd pb
                   on pb_id.customer_id = pb.customer_id
WHERE cd.application_date >= '2025-10-07';






DROP TABLE IF EXISTS #temp_sfs_customer_credit_attributes;
SELECT cr.customer_id
     , CASE
           WHEN MAX(CASE WHEN (attribute_name = 'P_AUT_DQ001') THEN attribute_value END) > 0 THEN 1
           ELSE 0 END                                                                                AS dq_auto
     , cast(MAX(Coalesce(CASE
                             WHEN (attribute_name = 'Score_5402_1') and attribute_value > 300 THEN attribute_value
                             else null END,
                CASE WHEN (attribute_name = 'ConsumerCreditScore') THEN attribute_value END)) as int) AS vantage
     , cast(MAX(CASE WHEN (attribute_name = 'Score_5207_1') THEN attribute_value END) as int)        AS fico
     , COALESCE((MAX(CASE WHEN (attribute_name = 'P_CCS_EX001') THEN attribute_value::int END) > 0)::int,
                0)                                                                                   AS secured_credit_card


--      , COALESCE((MAX(CASE WHEN (attribute_name = 'ChimeIndicator') THEN attribute_value::int END) > 0)::int, 0)                                                                            AS old_chime_indicator
--      , coalesce((max(chime_table.chime_cards) > 0)::int, 0)                                   as new_chime


-- max of chime flag calculated two different way as a failsafe
      , case
            when COALESCE((MAX(CASE WHEN (attribute_name = 'ChimeIndicator') THEN attribute_value::int END) >
                           0)::int,
                          0) > coalesce((max(chime_table.chime_cards) > 0)::int, 0) then COALESCE(
                    (MAX(CASE WHEN (attribute_name = 'ChimeIndicator') THEN attribute_value::int END) > 0)::int,
                    0)
            else coalesce((max(chime_table.chime_cards) > 0)::int, 0) end                             as chime_indicator


     , COALESCE(MAX(CASE WHEN (attribute_name IN ('SUCI136', 'C_CIN_EX001')) THEN attribute_value::int END),
                0)                                                                                   AS num_tradelines


     , COALESCE(MAX(CASE WHEN (attribute_name IN ('SUC2062', 'C_CIN_EX013')) THEN attribute_value::int END),
                0)                                                                                   AS auth_tradelines
     , COALESCE(MAX(CASE WHEN (attribute_name = 'ACAChargeoff') THEN attribute_value::int END),
                0)                                                                                   AS prev_chargeoff
     , COALESCE(MAX(CASE WHEN (attribute_name IN ('C_CIN_EX002')) THEN attribute_value::int END),
                0)                                                                                   AS open_tradelines
INTO #temp_sfs_customer_credit_attributes
FROM odsnpi.sfs_customer_report_scd cr
         left join
     odsnpi.sfs_customer_report_attribute_scd cra
     on cra.customer_report_id = cr.customer_report_id and cra.current_version_flag = 1 and cr.current_version_flag = 1

--could be outdated, add data source id,


          left join(select cr.customer_id,
                           max(case when brt.member_name ilike '%chime%' then 1 else 0 end) as chime_cards

                    from custrpt.report_index ri
                             left join custrpt.bureau_report_tradeline brt
                                       on ri.report_key = brt.report_key
                             left join odsnpi.los_customer_report_scd cr
                                       on ri.report_key = cr.external_report_id

                    where ri.product_name ilike '%dms%'
                    group by 1) as chime_table
                   on cr.customer_id = chime_table.customer_id


GROUP BY cr.customer_id;

DROP TABLE IF EXISTS #temp_blackbook_values;
WITH btr_trimmed_json_table AS (SELECT loan_id,
                                       CASE
                                           WHEN MAX(created_utc_dtm) <= '2023-05-22' THEN 0.88 * MAX(blackbook_value)
                                           ELSE MAX(blackbook_value) END                                                                            AS bb_value_new,
                                       TRIM('"' FROM
                                            JSON_SERIALIZE(blackbook_template16_response_json."used_vehicles"."used_vehicle_list"[0]."class_name")) AS car_class_new,
                                       TRIM('"' FROM
                                            JSON_SERIALIZE(blackbook_template16_response_json."used_vehicles"."used_vehicle_list"[0]."fuel_type"))  AS car_fuel_new
                                FROM sandbox.blackbook_template16_reports
                                WHERE blackbook_value IS NOT NULL
                                  AND blackbook_value != 0
                                GROUP BY loan_id, car_class_new, car_fuel_new)
   , un_duplicated_kmx_blackbook AS (SELECT account_number, AVG(blackbook_value) AS bb_value
                                     FROM sandbox.kmx_blackbook
                                     GROUP BY account_number)
   , btr_trimmed_json_table_kmx AS (select distinct btrim(upper(bb.vin)) as vin,
                                                    veh_class            as car_class_new,
                                                    veh_fuel             as car_fuel_new,
                                                    case
                                                        when veh_fuel ilike '%Gas%' then 1
                                                        when veh_fuel = 'Flex' then 2
                                                        when veh_fuel ilike '%Hybrid%' then 3
                                                        when veh_fuel = 'Diesel' then 4
                                                        else 5
                                                        end              as fuel_prio
                                    from (select vin,
                                                 trim('\\' from blackbook_template21_response_json::varchar)            as json_clean,
                                                 regexp_substr(json_clean,
                                                               '"class_name":"(\\w+[ -/]?\\w*[ -/]?\\w*[ -/]?\\w*)","class_name_long"',
                                                               1, 1,
                                                               'e')                                                     as veh_class,
                                                 regexp_substr(json_clean, '"fuel_type":"((\\w+[ -/]?)*)",', 1, 1, 'e') as veh_fuel
                                          from sandbox.shiny_blackbook_template21_reports) bb
                                    where veh_class is not null
                                      and veh_class not ilike '%null%'
                                      and veh_class != ''
                                      and veh_fuel is not null
                                      and veh_fuel not ilike '%null%'
                                      and veh_fuel != '')
SELECT distinct cd.account_number
              , COALESCE(kb.bb_value, hkb.wholesale_average_adjusted, bb22_23.used_car_wholesale_average_adjusted,
                         btr.bb_value_new, rbov.wholesale_average_adjusted)                              AS bb_wholesale
              , COALESCE(bbi.vehicle_class_name,
                         bb22_23.used_car_vehicle_class_name, btr.car_class_new, bbk2.car_class_new)     AS car_class
              , COALESCE(bbi.extended_vehicle_class_name,
                         bb22_23.used_car_extended_vehicle_class_name)                                   AS car_class_ext
              , COALESCE(bbi.fuel_type, bb22_23.used_car_fuel_type, btr.car_fuel_new, bbk2.car_fuel_new) AS car_fuel
INTO #temp_blackbook_values
FROM edwnpi.los_deal_current_fact AS cd
         LEFT JOIN sandbox.rds_blackbook_origination_values AS rbov -- Pre-2023 Q1, all LOB (values)
                   ON cd.account_number = rbov.account_number
                       AND cd.purchase_odometer = rbov.mileage_at_origination
         LEFT JOIN sandbox.rds_blackbook_vehicle_info AS bbi -- Pre-2023 Q1, all LOB (info)
                   ON cd.account_number = bbi.account_number
         LEFT JOIN sandbox.historical_blackbook_q2_2022_q2_2023 AS bb22_23 -- 2022 Q1-2023 Q2, all LOB (values and info)
                   ON cd.loan_id = bb22_23.appid
         LEFT JOIN btr_trimmed_json_table AS btr -- 2023 Q2+, nonKMX (values and JSON)
                   ON cd.loan_id = btr.loan_id
         LEFT JOIN un_duplicated_kmx_blackbook AS kb
                   ON cd.account_number = kb.account_number
         LEFT JOIN btr_trimmed_json_table_kmx AS bbk2
                   on bbk2.vin = cd.purchase_vin
         LEFT JOIN sandbox.historical_KMX_blackbook AS hkb -- Assortment of KMX 2018 Q2 to 2023 Q4
                   ON cd.loan_id = hkb.application_id
         LEFT JOIN edwnpi.dealer_rollup_scd_current AS dru
                   ON cd.dealer_number = dru.dealer_number
WHERE cd.account_number is NOT NULL;



DROP TABLE IF EXISTS #temp_fraud;

with sentilink_table AS (select loan_id,
                                ldcf.application_received_date,

                                case
                                    when pb_sentilink_first_party_synthetic_score between 800 and 999
                                        then 2.7 --'sl800-999'
                                    when pb_sentilink_first_party_synthetic_score between 700 and 800 then 1.23--'sl700-800'
                                    else 0.0
                                    end as sentilink_adjustment


                         from acaedw.edwnpi.los_deal_current_fact ldcf
                                  left join acaedw.sandbox.pos_fraud_data pfd on pfd.application_id = ldcf.loan_id
                         where aspect ilike 'contract'
                           and ldcf.data_source_id in (17, 100)
                           and ldcf.application_received_date >= '2022-11-08'
                           and ldcf.status_name = 'FUNDED'
                           and deal_deleted_flag = 0),


     pointpredictive_table as (select loan_id,

                                      case
                                          when afm_score = 999 then 1.34 --'pp999'
                                          when afm_score between 750 and 998 then 1.38 --'pp750-998'
                                          when afm_score between 600 and 750 then 1.28 --'pp600-750'
                                          else 0.0 end as point_predictive_adjustment


                               from acaedw.edwnpi.los_deal_current_fact ldcf
                                        left join acaedw.sandbox.pos_fraud_data pfd on pfd.application_id = ldcf.loan_id
                               where aspect ilike 'contract'
                                 and ldcf.data_source_id in (17, 100)
                                 and ldcf.application_received_date >= '2022-11-08'
                                 and ----'2024-05-31' and
                                   ldcf.status_name = 'FUNDED'
                                 and deal_deleted_flag = 0
                                 and (pfd.reason_code_1 in
                                      (8, 13, 17, 31, 32, 35, 50, 55, 59, 60, 61, 62, 63, 64, 65, 66, 67) or
                                      pfd.reason_code_2 in
                                      (8, 13, 17, 31, 32, 35, 50, 55, 59, 60, 61, 62, 63, 64, 65, 66, 67) or
                                      pfd.reason_code_3 in
                                      (8, 13, 17, 31, 32, 35, 50, 55, 59, 60, 61, 62, 63, 64, 65, 66, 67)) --and
--       pfd.pb_sentilink_first_party_synthetic_score < 700
     ),


     low_fraud_table AS (select ldcf.loan_id
--      ,ldcf.application_received_date
--      ,substring(cdd.pricing_hurdle_name, 6, 8) as dealergroup
--      ,pfd.afm_score
--      ,pfd.pb_sentilink_first_party_synthetic_score
--      ,coalesce(pds.amtfinancedback, ldcf.con_amount_financed_back) as atf
-- --      ,dss.tier_model_score
-- --      ,dss.base_model_score
-- --      ,apppdsm.modelvalue
--      ,coalesce(pds.ltvfront,ldcf.con_ltv_front) as ltv


                              , case
                                    when (substring(cdd.pricing_hurdle_name, 6, 8) ilike 'KMX'
                                        and pfd.afm_score <= 300
                                        and pfd.pb_sentilink_first_party_synthetic_score < 500
                                        and coalesce(pds.amtfinancedback, ldcf.con_amount_financed_back) < 40000
                                        and coalesce(pds.ltvfront, ldcf.con_ltv_front) < 1.3
                                        and coalesce(dss.tier_model_score, apppdsm.modelvalue) >= 131
                                        )
                                        then 0.0


                                    when (substring(cdd.pricing_hurdle_name, 6, 8) not ilike 'KMX'
                                        and pfd.afm_score <= 300
                                        and pfd.pb_sentilink_first_party_synthetic_score < 500
--                  pds.amtfinancedback < 40000 and
                                        and coalesce(pds.amtfinancedback, ldcf.con_amount_financed_back) < 40000
                                        and coalesce(pds.ltvfront, ldcf.con_ltv_front) < 1.3
                                        and coalesce(dss.base_model_score, apppdsm.modelvalue) >= 135
--                and pds.ltvfront < 1.3
                                        )
                                        then 0.0


                                    else 0 end as low_fraud_adjustment


                         from acaedw.edwnpi.los_deal_current_fact ldcf
                                  left join acaedw.edwnpi.los_deal_current_fact as app_ldcf
                                            on ldcf.loan_id = app_ldcf.loan_id
                                                and app_ldcf.aspect ilike 'application'
                                                and app_ldcf.data_source_id in (17, 100)
                                  left join acaedw.sandbox.pos_fraud_data as pfd
                                            on pfd.application_id = app_ldcf.loan_id
                                  left join acaedw.edwnpi.crm_dealer_dim as cdd
                                            on cdd.dealer_number = app_ldcf.dealer_number
                                                and cdd.current_version_flag = 1
                                  left join acaedw.odsnpi.los_deal_scenario_scd dss
                                            on dss.deal_detail_id = app_ldcf.los_deal_current_fact_universal_id
                                                and dss.active_flag = 1 and dss.current_version_flag = 1
                                                and scenario_type ilike 'Application'
                                  left join acaedw.odsnpi.prov_deal_scenarios as pds
                                            on pds.dealdetailid = app_ldcf.los_deal_current_fact_universal_id
                                                and pds.datasourceid = 17
                                                and pds.itemcd ilike 'Active'
                                  Left Join acaedw.odsnpi.prov_deal_scoremodels as apppdsm
                                            on apppdsm.scenarioid = pds.scenarioid
                                                and apppdsm.datasourceid = 17
                         where ldcf.loan_id is not null
                           and ldcf.aspect ilike 'contract'
                           and ldcf.data_source_id in (17, 100)
                           and ldcf.application_received_date >= '2022-11-08'
--  and ldcf.application_received_date < '2024-12-10'
                           and ldcf.deal_deleted_flag = 0
                           and ldcf.book_date is not null)


select sentilink_table.loan_id,
       sentilink_table.application_received_date,

       sentilink_adjustment,
       point_predictive_adjustment,
       low_fraud_adjustment,

       greatest(sentilink_adjustment, point_predictive_adjustment, low_fraud_adjustment) as fraud_adjustment


INTO #temp_fraud

from sentilink_table

         left join pointpredictive_table
                   on sentilink_table.loan_id = pointpredictive_table.loan_id
         left join low_fraud_table
                   on sentilink_table.loan_id = low_fraud_table.loan_id;
